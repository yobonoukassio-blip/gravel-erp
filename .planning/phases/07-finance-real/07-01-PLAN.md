    1. Create `apps/api/src/modules/analytics/jobs/cost-per-ton-aggregator.job.ts`:

       ```typescript
       import { Injectable, Logger } from '@nestjs/common';
       import { Cron, CronExpression } from '@nestjs/schedule';
       import { InjectDataSource } from '@nestjs/typeorm';
       import { DataSource } from 'typeorm';
       import { CostPerTonAggregatorService } from '../services/cost-per-ton-aggregator.service';

       @Injectable()
       export class CostPerTonAggregatorJob {
         private readonly logger = new Logger(CostPerTonAggregatorJob.name);

         constructor(
           @InjectDataSource() private readonly ds: DataSource,
           private readonly aggregator: CostPerTonAggregatorService,
         ) {}

         @Cron('0 4 * * *', { name: 'cost-per-ton-daily', timeZone: 'UTC' })
         async handleCron(): Promise<void> {
           const yesterday = new Date();
           yesterday.setUTCDate(yesterday.getUTCDate() - 1);
           const snapshotDate = yesterday.toISOString().split('T')[0];

           this.logger.log(`[CostPerTon] starting daily aggregation for ${snapshotDate}`);

           // Get all active tenants
           const tenants = await this.ds.query<Array<{ id: string }>>(
             `SELECT id FROM tenant WHERE is_active = true`,
           );

           for (const tenant of tenants) {
             await this.aggregateForTenant(tenant.id, snapshotDate);
           }

           this.logger.log(`[CostPerTon] completed for ${tenants.length} tenants`);
         }

         private async aggregateForTenant(tenantId: string, snapshotDate: string): Promise<void> {
           try {
             // Get all sites for this tenant
             const sites = await this.ds.query<Array<{ id: string }>>(
               `SELECT id FROM site WHERE tenant_id = $1 AND is_active = true`,
               [tenantId],
             );

             // Get all distinct calibre_codes with production
             const calibres = await this.ds.query<Array<{ calibre_code: string }>>(
               `SELECT DISTINCT calibre_code FROM stockpile
                WHERE tenant_id = $1 AND is_active = true AND calibre_code IS NOT NULL`,
               [tenantId],
             );

             for (const site of sites) {
               for (const calibre of calibres) {
                 try {
                   await this.aggregator.aggregateForDate({
                     tenantId,
                     siteId: site.id,
                     snapshotDate,
                     calibreCode: calibre.calibre_code,
                   });
                 } catch (err) {
                   this.logger.error(
                     `aggregate failed tenant=${tenantId} site=${site.id} calibre=${calibre.calibre_code}: ${(err as Error).message}`,
                   );
                 }
               }
             }
           } catch (err) {
             this.logger.error(`tenant ${tenantId} aggregation failed: ${(err as Error).message}`);
           }
         }
       }
       ```
